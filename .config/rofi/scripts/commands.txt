# Системные команды
 systemctl restart NetworkManager :: Перезапуск сети
 sudo pacman -Syu :: Обновление системы
 docker ps :: Список контейнеров
 htop :: Системный монитор
 neofetch :: Информация о системе

# Git команды
 git status :: Статус репозитория
 git pull :: Получить изменения
 git push :: Отправить изменения
 git add . :: Добавить все файлы
 git commit -m "" :: Создать коммит

# Файловые операции
 ls -la :: Список файлов
 df -h :: Информация о дисках
 free -h :: Информация о памяти
